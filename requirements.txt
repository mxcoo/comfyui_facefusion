# requirements.txt for ComfyUI-FaceFusion
# onnxruntime-gpu with CUDA support is required for GPU-accelerated face swapping

# Core dependency — install with CUDA support:
#   pip install "onnxruntime-gpu[cuda]==1.26.0"
#
# If you see CPU-only inference after installation, make sure the CPU-only
# `onnxruntime` package is NOT installed:
#   pip uninstall onnxruntime -y
#
# Then reinstall the GPU version:
#   pip install "onnxruntime-gpu[cuda]==1.26.0"
